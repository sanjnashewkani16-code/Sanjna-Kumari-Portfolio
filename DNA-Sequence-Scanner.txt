#include <iostream>

using namespace std;

bool validateDNA(string dna) {
    for (char c : dna) {
        if (c != 'A' && c != 'C' && c != 'G' && c != 'T') {
            return false;
        }
    }
    return true;
}

vector<vector<int>> buildDP(string X, string Y) {

    int m = X.length();
    int n = Y.length();

    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));

    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {

            if (X[i - 1] == Y[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            }

            else {
                dp[i][j] =
                max(dp[i - 1][j],
                    dp[i][j - 1]);
            }
        }
    }

    return dp;
}

string findLCS(string X, string Y,
vector<vector<int>>& dp) {

    int i = X.length();
    int j = Y.length();

    string lcs = "";

    while (i > 0 && j > 0) {

        if (X[i - 1] == Y[j - 1]) {

            lcs =
            X[i - 1] + lcs;

            i--;
            j--;
        }

        else if (
            dp[i - 1][j] >
            dp[i][j - 1]) {

            i--;
        }

        else {
            j--;
        }
    }

    return lcs;
}

void printTable(vector<vector<int>>& dp) {

    cout << "\nDP Matrix\n\n";

    for (int i = 0; i < dp.size(); i++) {

        for (int j = 0; j < dp[i].size(); j++) {

            cout
            << setw(3)
            << dp[i][j];
        }

        cout << endl;
    }
}

double similarityPercent(
int lcsLength,
int maxLength) {

    return (
    (double)lcsLength
    / maxLength)
    * 100;
}
